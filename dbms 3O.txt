-- 1. CREATE DEPARTMENT TABLE
CREATE TABLE Department (
    dept_id INT PRIMARY KEY,
    dept_name VARCHAR(50) NOT NULL
);

-- 2. CREATE PROJECT TABLE
CREATE TABLE Project (
    project_id INT PRIMARY KEY,
    project_name VARCHAR(100) NOT NULL
);

-- 3. CREATE EMPLOYEE TABLE
CREATE TABLE Employee (
    emp_id INT PRIMARY KEY,
    emp_name VARCHAR(50) NOT NULL,
    salary DECIMAL(10,2),
    dept_id INT,
    project_id INT,
    manager_id INT,
    FOREIGN KEY (dept_id) REFERENCES Department(dept_id),
    FOREIGN KEY (project_id) REFERENCES Project(project_id)
);

-- 9. DISPLAY ALL EMPLOYEES
SELECT * FROM Employee;

-- 10. EMPLOYEES EARNING MORE THAN 60000
SELECT *
FROM Employee
WHERE salary > 60000;

-- 11. DISPLAY EMPLOYEE ID, NAME, AND SALARY
SELECT emp_id, emp_name, salary
FROM Employee;

-- 12. AGGREGATE FUNCTIONS
SELECT
    COUNT(*) AS total_employees,
    AVG(salary) AS average_salary,
    MAX(salary) AS highest_salary,
    MIN(salary) AS lowest_salary,
    SUM(salary) AS total_salary
FROM Employee;

-- 13. COUNT EMPLOYEES IN EACH DEPARTMENT
SELECT
    dept_id,
    COUNT(*) AS employee_count
FROM Employee
GROUP BY dept_id
ORDER BY dept_id;

-- 14. DEPARTMENTS WITH AVERAGE SALARY ABOVE 55000
SELECT
    dept_id,
    AVG(salary) AS average_salary
FROM Employee
GROUP BY dept_id
HAVING AVG(salary) > 55000;

-- 15. CLASSIFY EMPLOYEES BY SALARY
SELECT
    emp_id,
    emp_name,
    salary,
    CASE
        WHEN salary >= 70000 THEN 'High'
        WHEN salary >= 55000 THEN 'Medium'
        ELSE 'Low'
    END AS salary_category
FROM Employee;

-- 16. SORT EMPLOYEES BY SALARY
SELECT emp_id, emp_name, salary
FROM Employee
ORDER BY salary DESC;

-- 17. DEPARTMENT-WISE EMPLOYEE AND SALARY DETAILS
SELECT
    d.dept_name,
    COUNT(e.emp_id) AS total_employees,
    AVG(e.salary) AS average_salary,
    SUM(e.salary) AS total_salary
FROM Department d
JOIN Employee e
    ON d.dept_id = e.dept_id
GROUP BY d.dept_id, d.dept_name
ORDER BY average_salary DESC;

-- 18. PROJECT-WISE EMPLOYEE COUNT
SELECT
    p.project_name,
    COUNT(e.emp_id) AS employee_count
FROM Project p
LEFT JOIN Employee e
    ON p.project_id = e.project_id
GROUP BY p.project_id, p.project_name
ORDER BY employee_count DESC;
